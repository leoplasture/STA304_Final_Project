% ═══════════════════ Slide 2 — The Problem ═══════════════════
\begin{frame}{The Problem: Trust Deficit in Research Agents}
  \begin{block}{Research Agents Have a Credibility Gap}
    LLM-based research agents (DeepScientist, OpenClaw) produce \textbf{professional-looking reports} with citations --- but users cannot answer three critical questions:
    \begin{enumerate}
      \item \textbf{Which conclusions are backed by real sources?} Citations are untraceable
      \item \textbf{Which are model inferences or hallucinations?} No automated detection exists
      \item \textbf{Which have no evidence at all?} Wording often stronger than evidence justifies
    \end{enumerate}
  \end{block}

  \begin{block}{Real-World Consequence}
    {\footnotesize In our baseline testing with unmodified DeepScientist: a single report contained \textbf{3 fabricated citations} that passed visual inspection. The agent confidently cited ``Smith et al. 2023'' for claims --- but that paper does not exist in Semantic Scholar's database of 200M+ papers.}
  \end{block}
\end{frame}

% ═══════════════════ Slide 3 — Why This Matters ═══════════════════
\begin{frame}{Why This Matters: The Hallucination Problem}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{The Scale of the Problem}
    \begin{itemize}
      \item LLMs generate \textbf{3--27\% fabricated references} in academic writing tasks (Walters \& Wilder, 2023; Bhatt, 2024)
      \item In scientific peer review, \textbf{up to 69\% of GPT-4 generated reviews} contain hallucinated citations (Liang et al., 2024)
      \item These fabrications are \textbf{indistinguishable from real citations} without external verification
      \item For research agents that autonomously generate proposals and ideas, this risk is amplified
    \end{itemize}

    \column{0.5\textwidth}
    \textbf{Our Response}
    \begin{itemize}
      \item \textbf{Every claim} extracted from a source paper is verified against a \textbf{real API} (not an LLM)
      \item \textbf{3-source fallback}: Semantic Scholar (200M+ papers) $\to$ arXiv (2.4M open-access) $\to$ Crossref (150M+ records)
      \item Results are \textbf{scored}, not estimated: verdict + confidence + evidence snippet
      \item Final report includes an \textbf{auditable evidence chain} that cross-references every citation
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{FactCheck Pipeline: 5 Phases, 6 Mandatory Tools}

      \textbf{Phase 0 --- PDF Parsing (A)}\\[2pt]
      PyPDF2 $\to$ regex ``REFERENCES'' $\to$ split by \texttt{[N]} / \texttt{[Author,Year]} $\to$ Unicode title quotes $\to$ truncation at 200K chars, 40 claims. Handles IEEE, APA, named-author.
    \end{minipage}}

    \vspace{0.1cm}
    \colorbox{blue!12}{\begin{minipage}{\dimexpr\linewidth-2\fboxsep\relax}
      \footnotesize
      \textbf{Phase 1 --- Verify (A) \hspace{0.2cm} \textit{323 calls / 6 quests}}\\[2pt]
      (1) Semantic Scholar $\to$ (2) arXiv Title API $\to$ (3) Crossref DOI. Returns verdict + confidence + evidence snippet.
    \end{minipage}}

    \vspace{0.1cm}
    \colorbox{green!18}{\begin{minipage}{\dimexpr\linewidth-2\fboxsep\relax}
      \footnotesize
      \textbf{Phase 2 --- Score (B) \hspace{0.2cm} \textit{7-branch RYG}}\\[2pt]
      \begin{tabular}{ll}
        supported conf $\ge$ 0.7 & \textcolor{greenRYG}{$\bullet$} Green (correct) \\
        contradicted conf $\ge$ 0.7 & \textcolor{redRYG}{$\bullet$} Red (wrong) \\
        all other combinations & \textcolor{yellowRYG}{$\bullet$} Yellow (uncertain) \\
      \end{tabular}
    \end{minipage}}

    \column{0.5\textwidth}
    \colorbox{green!14}{\begin{minipage}{\dimexpr\linewidth-2\fboxsep\relax}
      \footnotesize
      \textbf{Phase 3 --- Render (B)}\\[2pt]

    \colorbox{orange!14}{\begin{minipage}{\dimexpr\linewidth-2\fboxsep\relax}
      \footnotesize
      \textbf{Phase 4 --- Persist (C) \hspace{0.2cm} \textit{36/36 calls}}\\[2pt]
      \texttt{artifact\_\_record} $\to$ evidence\_store (auditable). \texttt{memory\_\_write} $\to$ episodes (traceable).
    \end{minipage}}

    \vspace{0.1cm}
    \colorbox{lightgray}{\begin{minipage}{\dimexpr\linewidth-2\fboxsep\relax}

  \end{columns}
\end{frame}

\begin{frame}{Evidence Chain: Making Every Claim Auditable}
  \begin{columns}[T]
    \column{0.38\textwidth}
    \textbf{3 Evidence Types (E00X)}
    \begin{table}[h]
      \footnotesize
      \begin{tabular}{ll}
        \toprule
        Source & ID Format \\
        \midrule
        QQ Text & E001 \\
        QQ Image & E001-img \\
        QQ PDF & E001-pdf \\
        \bottomrule
      \end{tabular}
    \end{table}
    {\scriptsize All entries: SHA256 hash. PDF: +page count, +text sidecar.}

    \vspace{0.05cm}
    \textbf{Report Structure}
    {\scriptsize §1 FactCheck, §2 Evidence Chain, §3--5 Bridges, §6 Caveats.}

    \column{0.59\textwidth}
    \textbf{Audit: \texttt{audit\_report()}}
    \begin{itemize}
      \footnotesize
      \item Scans report for evidence ID patterns (E, E-img, E-pdf)
      \item Cross-checks cited IDs vs.\ \texttt{evidence\_store}
      \item Detects fake IDs, unused evidence, bare claims
      \item Outputs PASS or WARN verdict
    \end{itemize}

    \vspace{0.1cm}
    \textbf{Three-Tier Annotation}
    \begin{description}
      \footnotesize
      \item[\texttt{[E001]}] Evidence-supported (\textcolor{greenRYG}{$\bullet$})
      \item[Inferred] Logically derived
      \item[Needs Verif.] Insufficient evidence
    \end{description}
  \end{columns}
\end{frame}

% ═══════════════════ Slide 10 — Failure Cases ═══════════════════
\begin{frame}{Failure Cases: What Breaks and How We Handle It}
  \begin{enumerate}
    \item \textbf{Verifier False Positives (LIGO Q031)}
    {\footnotesize C036 (GW190707 event) flagged \textcolor{redRYG}{$\bullet$} contradicted. Semantic Scholar matched a different paper with overlapping author names. \textbf{Handling}: Report explicitly states ``almost certainly a false negative --- verifier matched wrong paper.''}

    \item \textbf{Abstract-Only Verification Limits (SGD Q030)}
    {\footnotesize C018/C031 (mathematical convergence proofs) unverifiable from abstracts alone. \textbf{Handling}: Claims marked \textcolor{yellowRYG}{$\bullet$} uncertain; caveats note ``full-text access required for theorem verification.''}

    \item \textbf{PDF Parser Cross-Discipline Failure (Medical Q028)}
    {\footnotesize Medical papers indexed via PubMed, not arXiv --- \texttt{parse\_pdf} returned 0 structured claims despite successful text extraction. \textbf{Handling}: Agent auto-fell back to \texttt{bash\_exec} manual extraction; evidence table marked \texttt{extraction\_method: bash\_fallback}.}

    \item \textbf{DeepSeek API Version Incompatibility (early, fixed)}
    {\footnotesize Claude Code $\ge$2.1.154 sends \texttt{system} role messages unsupported by DeepSeek --- 80\% of runs crashed. \textbf{Fix}: locked Claude Code to v2.1.153, added \texttt{--strict-mcp-config}, filtered system Python from MCP PATH.}
  \end{enumerate}
\end{frame}

% ═══════════════════ Slide 14 — Limitations ═══════════════════
\begin{frame}{Limitations \& Future Work}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Current Limitations}
    \begin{itemize}
      \footnotesize
      \item Abstract-only: most \textcolor{yellowRYG}{$\bullet$} would be resolvable with full-text access
      \item Verifier false positives: $\sim$5--10\% \textcolor{redRYG}{$\bullet$} from title-matching errors
      \item PyPDF2 weak on scanned/image-based PDFs (requires OCR preprocessing)
      \item Network-dependent: offline mode not supported
      \item No video/audio evidence support
    \end{itemize}

    \column{0.5\textwidth}
    \textbf{Future Improvements}
    \begin{itemize}
      \footnotesize
      \item Full-text verification via arXiv open-access PDF parsing
      \item Local model fallback: Ollama + sentence-transformers for offline mode
      \item Structured NLP pipeline: replace regex with transformer-based claim extraction
      \item Multi-connector: extend to WeChat, Discord, Telegram
      \item Result caching: avoid duplicate API calls for same (claim, paper) pairs
    \end{itemize}
  \end{columns}
\end{frame}
