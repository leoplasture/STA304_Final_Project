\begin{frame}{The Problem}
  \begin{block}{Core Pain Point}
    Research Agents (DeepScientist, OpenClaw) generate \textbf{professional-looking reports}, but users cannot answer three questions:
    \begin{enumerate}
      \item \textbf{Which conclusions come from real sources?} --- citations untraceable
      \item \textbf{Which are model inferences?} --- no hallucination detection
      \item \textbf{Which have no evidence at all?} --- report wording stronger than evidence
    \end{enumerate}
  \end{block}

  \begin{block}{Our Goal}
    Build an \textbf{auditable, traceable evidence-chain system} that verifies every claim:
    \vspace{0.2cm}
    \begin{center}
      Upload PDF $\rightarrow$ Structured Evidence ID $\rightarrow$ API Verification $\rightarrow$ 🟢🟡🔴 Scoring $\rightarrow$ Auto Audit
    \end{center}
  \end{block}
\end{frame}

% Slide 4 — Pipeline
% ════════════════════════════════════════════════════════════════
\begin{frame}{FactCheck Pipeline Detail}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Phase 0 --- PDF Parsing}
    \begin{itemize}
      \footnotesize
      \item PyPDF2 text extraction
      \item Regex-based reference section detection
      \item Split by \texttt{[N]} or \texttt{[Author,Year]} markers
      \item Unicode quote matching for title extraction
      \item Output truncation (200K chars / 40 claims max)
    \end{itemize}

    \textbf{Phase 1 --- Claim Verification $\times N$}
    \begin{itemize}
      \footnotesize
      \item 3-tier fallback chain:
      \item 1. Semantic Scholar (title+abstract)
      \item 2. arXiv (Title API)
      \item 3. Crossref (DOI/title search)
      \item Output: verdict + confidence + evidence snippet
    \end{itemize}

    \column{0.5\textwidth}
    \textbf{Phase 2 --- Batch Scoring (B)}
    \begin{table}[h]
      \tiny
      \begin{tabular}{lll}
        \toprule
        Condition & Score & Signal \\
        \midrule
        supported + conf $\ge$ 0.7 & Green & $\textcolor{greenRYG}{\bullet}$ \\
        supported + conf $<$ 0.7 & Yellow & $\textcolor{yellowRYG}{\bullet}$ \\
        not\_found & Yellow & $\textcolor{yellowRYG}{\bullet}$ \\
        contradicted + conf $\ge$ 0.7 & Red & $\textcolor{redRYG}{\bullet}$ \\
        contradicted + conf $<$ 0.7 & Yellow & $\textcolor{yellowRYG}{\bullet}$ \\
        \bottomrule
      \end{tabular}
    \end{table}

    \textbf{Phase 3 --- Render (B)}
    \begin{itemize}
      \footnotesize
      \item Summary table (RYG counts)
      \item Per-claim detail cards
      \item Evidence Chain table (E00X $\rightarrow$ C00X)
    \end{itemize}

    \textbf{Phase 4 --- Persist (C)}
    \begin{itemize}
      \footnotesize
      \item artifact\_\_record $\rightarrow$ evidence\_store
      \item memory\_\_write $\rightarrow$ ``episodes'' kind
    \end{itemize}
  \end{columns}
\end{frame}

    \begin{table}[h]
      \small
      \begin{tabular}{lll}
        \toprule
        Type & ID & Metadata \\
        \midrule
        QQ Text & E001 & text + SHA256 \\
        QQ Image & E001-img & dims + SHA256 \\
        QQ PDF & E001-pdf & pages + SHA256 \\
        \bottomrule
      \end{tabular}
    \end{table}

    \vspace{0.2cm}
    \textbf{Data Flow}
    \begin{center}
      \tiny
      Claim (C001) $\rightarrow$ VerificationResult $\rightarrow$ ScoredClaimResult $\rightarrow$ FactCheckResult
    \end{center}

    \column{0.55\textwidth}
    \textbf{Dual-Layer Enforcement --- Agent Cannot Bypass}
    \begin{block}{Layer 1 --- SKILL.md State Machine}
      \footnotesize
      Score $\rightarrow$ Render $\rightarrow$ Record $\rightarrow$ Memory $\rightarrow$ Deliver

      Each state has exit checks; $\geq$1 fail = retry current state
    \end{block}

    \begin{block}{Layer 2 --- HARD STOP RULE (builder.py)}
      \footnotesize
      \texttt{MUST call score\_batch exactly once. No manual scoring.}

      \texttt{MUST call render\_report. No hand-written tables.}

      \texttt{MUST call artifact\_\_record, confirm status = ok.}

      \texttt{MUST call memory\_\_write after report generation.}

      \texttt{HARD STOP: if any tool missing, continue execution.}
    \end{block}
  \end{columns}
\end{frame}

% Slide 7 — Failure Cases

\begin{frame}{Failure Case Analysis}
  \begin{enumerate}
    \item \textbf{Verifier False Positive (LIGO Q031)}
    \begin{itemize}
      \footnotesize
      \item C036 (GW190707 event) flagged \textcolor{redRYG}{$\bullet$} contradicted
      \item \textbf{Cause}: Semantic Scholar matched a different paper with overlapping authors
      \item \textbf{Handling}: Report Caveats explicitly states ``almost certainly a false negative''
    \end{itemize}

    \item \textbf{Abstract-Only Limitation (SGD Q030)}
    \begin{itemize}
      \footnotesize
      \item C018/C031 mathematical claims unverifiable
      \item \textbf{Cause}: Theorems/lemmas require full-text, abstracts lack detail
    \end{itemize}

    \item \textbf{PDF Parser Fallback (Medical Q028)}
    \begin{itemize}
      \footnotesize
      \item \texttt{parse\_pdf} returned 0 claims; Agent auto-fallback to \texttt{bash\_exec}
      \item \textbf{Cause}: Medical papers indexed via PubMed, not arXiv --- different citation format
    \end{itemize}

    \item \textbf{Agent Bypassed Toolchain (mid-dev, fixed)}
    \begin{itemize}
      \footnotesize
      \item Early runs: only 10/40 verify\_call, \textbf{0 score\_batch}, \textbf{0 render\_report}
      \item \textbf{3-iteration fix}: MUST annotations $\rightarrow$ 5-State Machine $\rightarrow$ HARD STOP RULE
      \item \textbf{Result}: 36/36 compliance across all 6 quests
    \end{itemize}
  \end{enumerate}
\end{frame}

    \end{table}

    \column{0.45\textwidth}
    \textbf{Three-Tier Annotation}\\
    (Project core requirement)

    \begin{itemize}
      \item \texttt{[E001] [E030]} --- Claims \textbf{supported} by evidence ($\textcolor{greenRYG}{\bullet}$ green)
      \item \texttt{Inferred} --- \textbf{Logically inferred} but not directly proven by cited sources
      \item \texttt{Needs Verification} --- \textbf{Insufficient evidence}; requires further experiment or manual review
    \end{itemize}

    \vspace{0.3cm}
    \texttt{audit\_report()} automatically scans for these annotations and cross-references with \texttt{evidence\_store.json}.
  \end{columns}

\begin{frame}{Limitations \& Future Work}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Current Limitations}
    \begin{itemize}
      \footnotesize
      \item Abstract-only verification: most $\textcolor{yellowRYG}{\bullet}$ resolvable with full-text
      \item Verifier false positives: $\sim$5--10\% $\textcolor{redRYG}{\bullet}$ may be misclassifications
      \item PyPDF2 weak on scanned PDFs
      \item API-dependent: offline mode not supported
      \item No video/audio support yet
    \end{itemize}

    \column{0.5\textwidth}
    \textbf{Future Improvements}
    \begin{itemize}
      \footnotesize
      \item Full-text verification via arXiv open-access
      \item Local model fallback (Ollama + sentence-transformers)
      \item Structured NLP pipeline for claim extraction
      \item Multi-connector support (WeChat, Discord)
      \item Caching layer to avoid duplicate API calls
    \end{itemize}
  \end{columns}
\end{frame}
