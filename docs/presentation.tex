% Slide 2 — Problem
% ════════════════════════════════════════════════════════════════
\begin{frame}{The Problem}
  \begin{block}{Core Pain Point}
    Research Agents (DeepScientist, OpenClaw) generate \textbf{professional-looking reports}, but users cannot answer three questions:
    \begin{enumerate}
      \item \textbf{Which conclusions come from real sources?} --- citations untraceable
      \item \textbf{Which are model inferences?} --- no hallucination detection
      \item \textbf{Which have no evidence at all?} --- report wording stronger than evidence
    \end{enumerate}
  \end{block}

  \begin{block}{Our Goal}
    Build an \textbf{auditable, traceable evidence-chain system} that verifies every claim:
    \vspace{0.2cm}
    \begin{center}
      Upload PDF $\rightarrow$ Structured Evidence ID $\rightarrow$ API Verification $\rightarrow$ 🟢🟡🔴 Scoring $\rightarrow$ Auto Audit
    \end{center}
  \end{block}
\end{frame}

      \item[Person C --- Integration Layer] \texttt{crossdisc\_idea} 5-Phase Skill, QQ multi-modal evidence recording, daemon integration, memory schema
    \end{description}
  \end{block}

% Slide 4 — Pipeline
% ════════════════════════════════════════════════════════════════
\begin{frame}{FactCheck Pipeline Detail}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{Phase 0 --- PDF Parsing} \\[2pt]
    {\footnotesize PyPDF2 extraction\; $\to$\; regex detects ``REFERENCES'' heading\; $\to$\; split by \texttt{[N]} / \texttt{[Author,Year]}\; $\to$\; Unicode quote matching\; $\to$\; truncation (200K chars, 40 claims max)}

    \vspace{0.3cm}
    \textbf{Phase 1 --- Claim Verification} \\[2pt]
    {\footnotesize 3-tier fallback: (1) Semantic Scholar, (2) arXiv Title API, (3) Crossref DOI search. Output: verdict + confidence + evidence snippet.}

    \vspace{0.3cm}
    \textbf{Phase 2 --- Batch Scoring (B)} \\[2pt]
    {\footnotesize
    \begin{tabular}{ll}
      supported, conf $\ge$ 0.7 & $\textcolor{greenRYG}{\bullet}$ Green \\
      supported, conf $<$ 0.7 & $\textcolor{yellowRYG}{\bullet}$ Yellow \\
      not\_found, uncertain & $\textcolor{yellowRYG}{\bullet}$ Yellow \\
      contradicted, conf $\ge$ 0.7 & $\textcolor{redRYG}{\bullet}$ Red \\
      contradicted, conf $<$ 0.7 & $\textcolor{yellowRYG}{\bullet}$ Yellow \\
    \end{tabular}}

    \column{0.48\textwidth}
    \textbf{Phase 3 --- Render (B)} \\[2pt]
    {\footnotesize Rendered Markdown with: summary table (RYG counts), per-claim detail cards (verdict + confidence + evidence snippet), Evidence Chain table (E00X $\to$ C00X mapping).}

    \vspace{0.3cm}
    \textbf{Phase 4 --- Persist (C)} \\[2pt]
    {\footnotesize \texttt{artifact\_\_record} $\to$ evidence\_store \quad \texttt{memory\_\_write} $\to$ ``episodes'' kind.}

    \vspace{0.3cm}
    \textbf{Data Flow} \\[2pt]
    {\footnotesize Claim (C001) $\to$ VerificationResult $\to$ ScoredClaimResult $\to$ FactCheckResult}

    \vspace{0.3cm}
    \textbf{Enforcement --- Dual-Layer} \\[2pt]
    {\footnotesize
    Layer 1 (SKILL.md): 5-State Machine (Score$\to$Render$\to$Record$\to$Memory$\to$Deliver), each with exit checks.\\[2pt]
    Layer 2 (builder.py): HARD STOP RULE --- all mandatory tool calls must complete before answer finalization.}
  \end{columns}
\end{frame}

\begin{frame}{Evidence Chain \& Audit}
  \begin{columns}[T]
    \column{0.45\textwidth}
    \textbf{3 Evidence Entry Types}
    \begin{table}[h]
      \small
      \begin{tabular}{lll}
        \toprule
        Type & ID & Metadata \\
        \midrule
        QQ Text & E001 & text + SHA256 \\
        QQ Image & E001-img & dimensions + SHA256 \\
        QQ PDF & E001-pdf & pages + SHA256 + full-text \\
        \bottomrule
      \end{tabular}
    \end{table}

    \vspace{0.2cm}
    \textbf{Final Report Structure}
    \begin{itemize}
      \footnotesize
      \item §1 FactCheck Results (rendered RYG table)
      \item §2 Evidence Chain Table (E00X $\leftrightarrow$ C00X, 40 rows)
      \item §3--5 Cross-discipline bridges + hypothesis
      \item §6 Caveats (verifier limits, hedging language)
    \end{itemize}

    \column{0.55\textwidth}
    \textbf{Audit: \texttt{audit\_report()}}
    \begin{itemize}
      \footnotesize
      \item Scans report for \texttt{[E001]}, \texttt{[E001-img]}, \texttt{[E001-pdf]} patterns
      \item Cross-references every cited ID against \texttt{evidence\_store.json}
      \item Detects: fake IDs, unused evidence, bare claims (no annotation)
      \item Counts: supported / inferred / unverified claims
      \item Outputs PASS or WARN verdict
    \end{itemize}

    \vspace{0.2cm}
    \textbf{Three-Tier Annotation}
    \begin{description}
      \footnotesize
      \item[\texttt{[E001]}] Evidence-supported claim ($\textcolor{greenRYG}{\bullet}$ green)
      \item[Inferred] Logically inferred, not directly proven
      \item[Needs Verification] Insufficient evidence, needs further check
    \end{description}
  \end{columns}
\end{frame}

\begin{frame}{Failure Case Analysis}
  \begin{enumerate}
    \item \textbf{Verifier False Positive (LIGO Q031)}
    {\footnotesize C036 (GW190707) flagged \textcolor{redRYG}{$\bullet$} contradicted. Cause: S2 matched different paper with overlapping authors. Handling: Caveats states ``almost certainly a false negative.''}

    \item \textbf{Abstract-Only Limitation (SGD Q030)}
    {\footnotesize C018/C031 unverifiable --- theorems need full-text access, abstracts lack detail.}

    \item \textbf{PDF Parser Fallback (Medical Q028)}
    {\footnotesize \texttt{parse\_pdf} returned 0 claims (PubMed-indexed, not arXiv). Agent auto-fallback to \texttt{bash\_exec}. Extraction method marked as \texttt{bash\_fallback} in evidence table.}

    \item \textbf{Agent Bypassed Toolchain (mid-dev, \textbf{fixed})}
    {\footnotesize Early runs: 10/40 verify\_call, 0 score\_batch, 0 render\_report. Fix (3 iterations): MUST $\to$ 5-State Machine $\to$ HARD STOP RULE. Result: 36/36 compliance.}
  \end{enumerate}
\end{frame}

% Slide 10 — Limitations

\begin{frame}{Limitations \& Future Work}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Current Limitations}
    \begin{itemize}
      \footnotesize
      \item Abstract-only verification: most $\textcolor{yellowRYG}{\bullet}$ resolvable with full-text
      \item Verifier false positives: $\sim$5--10\% $\textcolor{redRYG}{\bullet}$ may be misclassifications
      \item PyPDF2 weak on scanned PDFs
      \item API-dependent: offline mode not supported
      \item No video/audio support yet
    \end{itemize}

    \column{0.5\textwidth}
    \textbf{Future Improvements}
    \begin{itemize}
      \footnotesize
      \item Full-text verification via arXiv open-access
      \item Local model fallback (Ollama + sentence-transformers)
      \item Structured NLP pipeline for claim extraction
      \item Multi-connector support (WeChat, Discord)
      \item Caching layer to avoid duplicate API calls
    \end{itemize}
  \end{columns}
\end{frame}
